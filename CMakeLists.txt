cmake_minimum_required(VERSION 3.16)
project(smc VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(smc STATIC main/ALG_SMC.cpp)
target_include_directories(smc PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/main)
target_compile_features(smc PUBLIC cxx_std_11)
set_target_properties(smc PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(smc PRIVATE -Wall -Wextra -Wpedantic -Werror
        -ffp-contract=off -fno-fast-math -fno-exceptions -fno-rtti)
endif()

function(smc_host_target name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE smc)
    set_target_properties(${name} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
    endif()
    if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_link_options(${name} PRIVATE -static)
    endif()
endfunction()

include(CTest)
if(BUILD_TESTING)
    foreach(name formula lifecycle invalid limits grid velocity_formula velocity_lifecycle velocity_invalid velocity_grid extensions_formula extensions_lifecycle extensions_invalid extensions_grid)
        smc_host_target(${name}_test tests/${name}_test.cpp)
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
endif()

option(SMC_BUILD_EXAMPLES "Build independent host examples and verification runners" OFF)
if(SMC_BUILD_EXAMPLES)
    smc_host_target(position_example examples/position_example.cpp)
    smc_host_target(verification_runner examples/verification_runner.cpp)
    smc_host_target(velocity_example examples/velocity_example.cpp)
    smc_host_target(velocity_verification_runner examples/velocity_verification_runner.cpp)
    smc_host_target(extensions_verification_runner examples/extensions_verification_runner.cpp)
endif()
